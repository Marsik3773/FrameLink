cmake_minimum_required(VERSION 3.24)
project(FrameLinkCamera LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BASECLASSES "" CACHE PATH "Microsoft DirectShow baseclasses source")
file(GLOB base_sources CONFIGURE_DEPENDS "${BASECLASSES}/*.cpp")
if(NOT EXISTS "${BASECLASSES}/streams.h")
  message(FATAL_ERROR "Pass -DBASECLASSES with the path to Microsoft's DirectShow baseclasses.")
endif()
add_library(dshow_base STATIC ${base_sources})
target_include_directories(dshow_base PUBLIC "${BASECLASSES}")
target_compile_definitions(dshow_base PUBLIC UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(dshow_base PRIVATE /W0)
add_library(FrameLinkCamera SHARED camera.cpp camera.def)
target_compile_features(FrameLinkCamera PRIVATE cxx_std_17)
target_compile_definitions(FrameLinkCamera PRIVATE NOMINMAX)
target_link_libraries(FrameLinkCamera PRIVATE dshow_base strmiids winmm ole32 oleaut32 uuid advapi32)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  add_executable(framelink-virtualcam ../virtualcam.c)
  target_compile_features(framelink-virtualcam PRIVATE c_std_17)
  target_link_libraries(framelink-virtualcam PRIVATE advapi32)
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  add_executable(framelink-camera-register ../mf/register.cpp)
  target_link_libraries(framelink-camera-register PRIVATE mf mfplat ole32 mfuuid)
endif()
